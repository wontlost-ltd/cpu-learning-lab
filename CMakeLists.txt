cmake_minimum_required(VERSION 3.16)
project(cpu_learning_lab LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(learning INTERFACE)
target_include_directories(learning INTERFACE include)
add_executable(cpu-learn apps/cli/main.cpp)
target_link_libraries(cpu-learn PRIVATE learning)
add_executable(cpu-sync apps/sync/main.cpp)
target_link_libraries(cpu-sync PRIVATE learning)
add_executable(cpu-listen apps/listen/main.cpp)
target_link_libraries(cpu-listen PRIVATE learning)
target_include_directories(cpu-listen PRIVATE adapters/audio)
option(CPUAI_WITH_NETWORK "Build optional POSIX TLS device transport (OpenSSL 3)" OFF)
if(CPUAI_WITH_NETWORK)
  if(WIN32)
    message(FATAL_ERROR "Network reference adapter currently supports macOS/Linux")
  endif()
  find_package(OpenSSL 3 REQUIRED)
  add_executable(cpu-link apps/link/main.cpp)
  target_include_directories(cpu-link PRIVATE adapters/network)
  target_link_libraries(cpu-link PRIVATE learning OpenSSL::SSL OpenSSL::Crypto)
endif()
option(CPUAI_WITH_PORTAUDIO "Enable native default microphone via installed PortAudio" OFF)
if(CPUAI_WITH_PORTAUDIO)
  find_path(PORTAUDIO_INCLUDE_DIR portaudio.h)
  find_library(PORTAUDIO_LIBRARY NAMES portaudio portaudio_static)
  if(NOT PORTAUDIO_INCLUDE_DIR OR NOT PORTAUDIO_LIBRARY)
    message(FATAL_ERROR "CPUAI_WITH_PORTAUDIO requires installed PortAudio headers and library")
  endif()
  target_include_directories(cpu-listen PRIVATE ${PORTAUDIO_INCLUDE_DIR})
  target_link_libraries(cpu-listen PRIVATE ${PORTAUDIO_LIBRARY})
  target_compile_definitions(cpu-listen PRIVATE CPUAI_WITH_PORTAUDIO=1)
endif()
option(CPUAI_BUILD_BENCHMARKS "Build the reproducible search benchmark" OFF)
if(CPUAI_BUILD_BENCHMARKS)
  add_executable(audio-benchmark benchmarks/audio_benchmark.cpp)
  target_link_libraries(audio-benchmark PRIVATE learning)
  add_executable(search-benchmark benchmarks/search_benchmark.cpp)
  target_link_libraries(search-benchmark PRIVATE learning)
endif()
include(CTest)
if(BUILD_TESTING)
  add_executable(network-protocol-tests tests/unit/network_protocol_test.cpp)
  target_link_libraries(network-protocol-tests PRIVATE learning)
  add_test(NAME network-protocol-tests COMMAND network-protocol-tests)
  add_executable(device-sync-tests tests/unit/device_sync_test.cpp)
  target_link_libraries(device-sync-tests PRIVATE learning)
  add_test(NAME device-sync-tests COMMAND device-sync-tests)
  add_executable(audio-stream-tests tests/unit/audio_stream_test.cpp)
  target_link_libraries(audio-stream-tests PRIVATE learning)
  add_test(NAME audio-stream-tests COMMAND audio-stream-tests)
  add_executable(observations-tests tests/unit/observations_test.cpp)
  target_link_libraries(observations-tests PRIVATE learning)
  add_test(NAME observations-tests COMMAND observations-tests)
  add_executable(learning-tests tests/unit/learner_test.cpp)
  target_link_libraries(learning-tests PRIVATE learning)
  add_executable(evaluation-tests tests/unit/evaluation_test.cpp)
  target_link_libraries(evaluation-tests PRIVATE learning)
  add_executable(skills-tests tests/unit/skills_test.cpp)
  target_link_libraries(skills-tests PRIVATE learning)
  add_test(NAME skills-tests COMMAND skills-tests)
  add_executable(runtime-tests tests/unit/runtime_test.cpp)
  target_link_libraries(runtime-tests PRIVATE learning)
  add_test(NAME runtime-tests COMMAND runtime-tests)
  add_test(NAME learning-tests COMMAND learning-tests)
  add_test(NAME evaluation-tests COMMAND evaluation-tests ${CMAKE_CURRENT_SOURCE_DIR}/tests/evaluation/square-v1.exam)
  find_package(Python3 COMPONENTS Interpreter REQUIRED)
  if(CPUAI_WITH_NETWORK)
    add_test(NAME network-cli-tests COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/integration/network_cli_test.py $<TARGET_FILE:cpu-link> $<TARGET_FILE:cpu-sync>)
    set_tests_properties(network-cli-tests PROPERTIES TIMEOUT 90)
  endif()
  add_test(NAME sync-cli-tests COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/integration/sync_cli_test.py $<TARGET_FILE:cpu-sync> $<TARGET_FILE:cpu-listen>)
  set_tests_properties(sync-cli-tests PROPERTIES TIMEOUT 60)
  add_test(NAME listen-cli-tests COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/integration/listen_cli_test.py $<TARGET_FILE:cpu-listen> $<TARGET_FILE:cpu-learn>)
  set_tests_properties(listen-cli-tests PROPERTIES TIMEOUT 45)
  add_test(NAME audio-cli-tests COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/integration/audio_cli_test.py $<TARGET_FILE:cpu-learn>)
  add_test(NAME skills-cli-tests COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/integration/skills_cli_test.py $<TARGET_FILE:cpu-learn> ${CMAKE_CURRENT_SOURCE_DIR}/tests/evaluation)
  add_test(NAME cli-tests COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/integration/cli_test.py $<TARGET_FILE:cpu-learn> ${CMAKE_CURRENT_SOURCE_DIR}/tests/evaluation/square-v1.exam)
endif()

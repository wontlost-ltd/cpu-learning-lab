cmake_minimum_required(VERSION 3.16)
project(cpu_learning_lab LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(learning INTERFACE)
target_include_directories(learning INTERFACE include)
add_executable(cpu-learn apps/cli/main.cpp)
target_link_libraries(cpu-learn PRIVATE learning)
add_executable(learning-tests tests/unit/learner_test.cpp)
target_link_libraries(learning-tests PRIVATE learning)
enable_testing()
add_test(NAME learning-tests COMMAND learning-tests)
